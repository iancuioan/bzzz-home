from django.db import models
from django.contrib.auth.models import User
from django.core.validators import MinValueValidator, MaxValueValidator
from django.core.exceptions import ValidationError
from datetime import date

# Create your models here.

def validate_leader_year(value):
    today = date.today()
    five_years_ago = today.replace(year=today.year - 5)
    if value > today:
        raise ValidationError("Data nu poate fi în viitor.")
    if value < five_years_ago:
        raise ValidationError("Data trebuie să fie din ultimii 5 ani.")

class House(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE, related_name='houses')
    name = models.CharField(max_length=100)
    #description = models.TextField(blank=True, null=True)

    notes = models.TextField(blank=True, null=True)
    #house_history = models.TextField(blank=True, null=True)  # nou: jurnal intern
    rating = models.IntegerField(
        validators=[MinValueValidator(1), MaxValueValidator(5)],
        default=1,
        help_text="Rating de la 1 la 5 stele"
    )

    class Meta:
        unique_together = ('user', 'name')
        ordering = ['-leader_year']
